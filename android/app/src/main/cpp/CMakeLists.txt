cmake_minimum_required(VERSION 3.22.1)
project(inpxbridge)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(NODE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../libnode")
add_library(node SHARED IMPORTED)
set_target_properties(node PROPERTIES IMPORTED_LOCATION "${NODE_DIR}/bin/${ANDROID_ABI}/libnode.so")
add_library(inpxbridge SHARED bridge.cpp)
target_include_directories(inpxbridge PRIVATE "${NODE_DIR}/include/node")
target_link_libraries(inpxbridge node log)
target_link_options(inpxbridge PRIVATE "-Wl,-z,max-page-size=16384")
